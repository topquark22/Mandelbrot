\newtheorem*{theorem}{Theorem}

\begin{theorem}[Escape radius and eventual exponential growth]
Let \(f(z)=z^q+c\) with \(q>1\), and define the orbit \(z_{n+1}=f(z_n)\) with \(z_0=0\).
Set
\[
R := 2^{1/(q-1)} \quad \text{so that} \quad R^{q-1}=2.
\]
Then:
\begin{enumerate}
\item If \(|c|>R\), then \(|z_n|\to\infty\).
\item If \(|c|\le R\) and there exists \(n\) with \(|z_n|>R\), then \(|z_n|\to\infty\).
Moreover, in this case there exists \(M\ge R\) (depending only on \(q\)) such that whenever \(|z_n|\ge M\) we have
\[
|z_{n+1}| \ge 2|z_n|.
\]
\end{enumerate}
\end{theorem}

\begin{proof}
Write \(x_n := |z_n|\). The reverse triangle inequality gives, for all \(n\),
\begin{equation}\label{eq:basic}
x_{n+1} = |z_{n+1}| = |z_n^q + c| \ge |z_n|^q - |c| = x_n^q - |c|.
\end{equation}

\textbf{Case 1: \(|c|>R\).}
Here \(z_1=c\), so \(x_1=|c|>R\). Then
\[
x_2 = |c^q + c| \ge |c|^q - |c| = |c|\bigl(|c|^{q-1}-1\bigr) > |c| = x_1,
\]
since \(|c|^{q-1} > R^{q-1}=2\).
Now assume inductively that \(x_n \ge |c|\) for some \(n\ge 1\). Then \eqref{eq:basic} implies
\[
x_{n+1} \ge x_n^q - |c| \ge x_n^q - x_n = x_n\bigl(x_n^{q-1}-1\bigr) > x_n,
\]
because \(x_n \ge |c|>R\) gives \(x_n^{q-1} > 2\).
Thus \((x_n)\) is strictly increasing for all \(n\ge 1\).

To conclude divergence, choose
\[
M_c := \max\{\,|c|,\ (2|c|)^{1/q}\,\}.
\]
Once \(x_n\ge M_c\), we have \(x_n^q \ge 2|c|\), hence \eqref{eq:basic} gives
\[
x_{n+1} \ge x_n^q - |c| \ge \frac{x_n^q}{2}.
\]
Iterating this inequality forces \(x_n\to\infty\), and therefore \(|z_n|=x_n\to\infty\) when \(|c|>R\).

\medskip
\noindent
\textbf{Case 2: \(|c|\le R\).}
Assume there exists \(n\) with \(x_n>R\). Then using \eqref{eq:basic} and \(|c|\le R\),
\[
x_{n+1} \ge x_n^q - R
= x_n\!\left(x_n^{q-1}-\frac{R}{x_n}\right)
> x_n(2-1)=x_n,
\]
because \(x_n^{q-1}>R^{q-1}=2\) and \(R/x_n<1\). Hence once the orbit crosses \(R\), the sequence \((x_n)\) is strictly increasing.

Now choose \(M\ge R\) so that
\[
M^{q-1}\ge 4 \qquad\text{and}\qquad M^q \ge 2R.
\]
(For example, \(M:=\max\{4^{1/(q-1)},(2R)^{1/q}\}\) works.)
Then for every \(x\ge M\), we have \(R \le x^q/2\), and therefore
\[
x^q - R \ge \frac{x^q}{2} \ge 2x,
\]
where the last inequality uses \(x^{q-1}\ge 4\). Applying this with \(x=x_n\) and using \eqref{eq:basic} with \(|c|\le R\) yields:
whenever \(x_n\ge M\),
\begin{equation}\label{eq:doubling}
x_{n+1} \ge x_n^q - R \ge 2x_n.
\end{equation}

Since \((x_n)\) is strictly increasing after it first exceeds \(R\), it must eventually exceed \(M\); let \(n_0\) be the first index with \(x_{n_0}\ge M\).
Then \eqref{eq:doubling} implies \(x_{n_0+m}\ge 2^m x_{n_0}\to\infty\) as \(m\to\infty\), proving \(x_n\to\infty\) and hence \(|z_n|\to\infty\).
\end{proof}
