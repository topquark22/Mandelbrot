Example usage from your existing Main

// ... package gtf.mandelbrot;
import gtf.mandelbrot.gpu.MandelbrotCuda;
import java.awt.image.BufferedImage;
import javax.imageio.ImageIO;
import java.io.File;

public class Main {
    public static void main(String[] args) throws Exception {
        int width = 1920, height = 1080, maxIter = 1000;
        double xmin = -2.0, xmax = 1.0, ymin = -1.5, ymax = 1.5;

        MandelbrotCuda gpu = new MandelbrotCuda("native/mandelbrot.ptx");
        int[] iters = gpu.compute(width, height, xmin, xmax, ymin, ymax, maxIter);
        gpu.close();

        BufferedImage img = new BufferedImage(width, height, BufferedImage.TYPE_INT_ARGB);
        for (int y=0; y<height; y++) {
            for (int x=0; x<width; x++) {
                int iter = iters[y*width + x];
                int argb = (iter==maxIter) ? 0xFF000000 : hsv(iter, maxIter);
                img.setRGB(x, y, argb);
            }
        }
        ImageIO.write(img, "png", new File("out.png"));
    }

    private static int hsv(int iter, int maxIter) {
        float h = (iter % maxIter) / (float) maxIter;
        float s = 0.7f, v = 1.0f;
        int rgb = java.awt.Color.HSBtoRGB(h, s, v);
        return 0xFF000000 | (rgb & 0x00FFFFFF);
    }
}

Ant integration
1) Add JCuda JARs

Download JCuda matching your installed CUDA Toolkit (Windows/Linux/macOS) and drop the JARs into lib/. (Version must match your CUDA driver/toolkit—check the site for the exact matrix.) 
jcuda.org

Classpath snippet in build.xml:

<path id="cp">
  <fileset dir="lib">
    <include name="**/*.jar"/>
  </fileset>
</path>

<target name="compile">
  <mkdir dir="bin"/>
  <javac srcdir="src" destdir="bin" includeantruntime="false">
    <classpath refid="cp"/>
  </javac>
</target>

JCuda provides OS-specific native binaries alongside the Java JARs. Place the native .dll/.so/.dylib where JCuda can extract or load them (same lib/ folder works with the “natives” JARs; otherwise add -Djava.library.path=lib). The official JCuda sample shows the full PTX-loading pattern we’re using. 
jcuda.org

2) Build the PTX with nvcc

Add a target to compile the kernel:

<property name="nvcc" value="nvcc"/>
<target name="ptx">
  <mkdir dir="native"/>
  <exec executable="${nvcc}" failonerror="true">
    <arg line="-ptx native/mandelbrot.cu -o native/mandelbrot.ptx"/>
    <!-- Optional arch tuning: -arch=compute_86 -->
  </exec>
</target>

<target name="run" depends="compile,ptx">
  <java classname="gtf.mandelbrot.Main" fork="true">
    <jvmarg value="-Djava.library.path=lib"/>
    <classpath>
      <path refid="cp"/>
      <pathelement location="bin"/>
    </classpath>
  </java>
</target>
